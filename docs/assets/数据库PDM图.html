系统 PDM 物理数据模型

app_user

  • PK id bigint
  • username varchar(32) UK
  • password varchar(128)
  • nickname varchar(32)
  • created_at datetime

user_token

  • PK id bigint
  • user_id bigint
  • token varchar(64) UK
  • created_at datetime
  • expires_at datetime

favorite

  • PK id bigint
  • user_id bigint
  • flight_id bigint
  • created_at datetime
  • UK(user_id, flight_id)

flight

  • PK id bigint
  • flight_no varchar(20)
  • airline_name varchar(64)
  • from_city / to_city varchar(32)
  • depart_time / arrive_time datetime
  • price decimal(10,2)
  • seats_left int
  • data_source varchar(32)

flight_price_snapshot

  • PK id bigint
  • flight_id bigint
  • flight_no varchar(20)
  • route fields
  • price decimal(10,2)
  • observed_at datetime

crawl_job

  • PK id bigint
  • status varchar(16)
  • started_at datetime
  • finished_at datetime
  • success_count int
  • failed_count int

flight_validation_failure

  • PK id bigint
  • crawl_job_id bigint
  • flight_no varchar(20)
  • failure_reason varchar(500)
  • rejected_at datetime

price_context

  • PK id bigint
  • from_city / to_city varchar(32)
  • depart_date date
  • context_text text
  • context_type varchar(32)

conversation_session

  • PK id varchar(36)
  • title varchar(128)
  • created_at datetime
  • updated_at datetime

conversation_message

  • PK id bigint
  • session_id varchar(36)
  • role varchar(16)
  • content text
  • created_at datetime
